const parms = new URLSearchParams(location.search); const id = parms.get("id"); const updateBoardBtn = document.getElementById("update-board"); const deleteBoardBtn = document.getElementById("delete-board"); const likeBtn = document.getElementById("like-btn"); const dislikeBtn = document.getElementById("dislike-btn"); // 🔑 게시글 ID별 고유 로컬스토리지 키 const storageKey = `reacted_board_${id}`; const countKey = `board_counts_${id}`; console.log("상세 조회 id:", id); // 1. 상세 게시글 내용 불러오기 fetch("/api/boards/" + id) .then(response => response.json()) .then(board => { document.getElementById("detail-title").value = board.title; document.getElementById("detail-writer").value = board.writer; document.getElementById("detail-content").value = board.content; // 💡 [핵심] 브라우저에 저장된 카운트가 있다면 그걸 쓰고, 없으면 서버 데이터(또는 0)를 사용 const localData = JSON.parse(localStorage.getItem(countKey)); let likeCount = localData ? localData.like : (board.likeCount || 0); let dislikeCount = localData ? localData.dislike : (board.dislikeCount || 0); updateCountsUI(likeCount, dislikeCount); restoreUserReactionUI(); }) .catch(error => { // API 조회 실패 시 로컬 데이터로 복원 const localData = JSON.parse(localStorage.getItem(countKey) || '{"like": 0, "dislike": 0}'); updateCountsUI(localData.like, localData.dislike); restoreUserReactionUI(); }); updateBoardBtn.addEventListener("click", updateBoard); deleteBoardBtn.addEventListener("click", deleteBoard); likeBtn.addEventListener("click", function() { sendReaction("like"); }); dislikeBtn.addEventListener("click", function() { sendReaction("dislike"); }); function sendReaction(type) { const previousReaction = localStorage.getItem(storageKey); if (previousReaction === type) { alert("이미 선택하셨습니다."); return; } // 현재 화면의 숫자를 가져옴 let likeCount = parseInt(document.getElementById("like-count").innerText) || 0; let dislikeCount = parseInt(document.getElementById("dislike-count").innerText) || 0; // 카운트 증감 로직 계산 if (previousReaction === "like" && type === "dislike") { likeCount = Math.max(0, likeCount - 1); dislikeCount += 1; } else if (previousReaction === "dislike" && type === "like") { dislikeCount = Math.max(0, dislikeCount - 1); likeCount += 1; } else { if (type === "like") likeCount += 1; if (type === "dislike") dislikeCount += 1; } // 💡 1. 화면 즉시 갱신 updateCountsUI(likeCount, dislikeCount); // 💡 2. 브라우저 localStorage에 강제 저장 (목록에 다녀와도 유지를 보장) localStorage.setItem(storageKey, type); localStorage.setItem(countKey, JSON.stringify({ like: likeCount, dislike: dislikeCount })); restoreUserReactionUI(); // 💡 3. 서버 전송 (서버 연동 시) fetch(`/api/boards/${id}/${type}`, { method: "POST" }).catch(err => console.log("서버 전송 생략/오류")); } // 화면 숫자 변경 함수 function updateCountsUI(like, dislike) { document.getElementById("like-count").innerText = like; document.getElementById("dislike-count").innerText = dislike; } // 눌렀던 버튼 파란색/빨간색 표시 복원 함수 function restoreUserReactionUI() { const currentReaction = localStorage.getItem(storageKey); likeBtn.classList.remove("active"); dislikeBtn.classList.remove("active"); if (currentReaction === "like") { likeBtn.classList.add("active"); } else if (currentReaction === "dislike") { dislikeBtn.classList.add("active"); } } function updateBoard() { const title = document.getElementById("detail-title").value; const writer = document.getElementById("detail-writer").value; const content = document.getElementById("detail-content").value; fetch("/api/boards/" + id, { method: "PUT", headers: { "Content-Type": "application/json" }, body: JSON.stringify({ title, writer, content }) }) .then(response => { if(response.ok) { alert("수정 완료"); location.href = "/board/list.html"; } }); } function deleteBoard() { if(!confirm("정말 삭제하시겠습니까?")) return; fetch("/api/boards/" + id, { method: "DELETE" }) .then(response => { if(response.ok) { alert("삭제 완료"); location.href = "/board/list.html"; } }); }
